HUSKYLENS huskylens;

// MP3 Pins: TX to 3, RX to 2
#define ARDUINO_RX 3  
#define ARDUINO_TX 2  
SerialMP3Player mp3(ARDUINO_RX, ARDUINO_TX);

int lastPlayedID = -1;

void setup() {
  // 1. Start Serial IMMEDIATELY
  Serial.begin(9600);
  delay(1000); 
  Serial.println("--- SERIAL MONITOR IS AWAKE ---");

  // 2. Initialize MP3 (Do this before Husky so it doesn't freeze)
  mp3.begin(9600);
  delay(1000);
  mp3.sendCommand(CMD_SEL_DEV, 0, 2); 
  delay(500);
  mp3.setVol(255);
  Serial.println("MP3 Player: Initialized & Volume Set to 30.");

  // 3. Initialize HuskyLens (Non-blocking)
  Wire.begin();
  Serial.println("Looking for HuskyLens...");
  if (!huskylens.begin(Wire)) {
    Serial.println("WARNING: HuskyLens not found. Check I2C setting & A4/A5.");
  } else {
    Serial.println("SUCCESS: HuskyLens Connected.");
  }

  Serial.println("--- READY: WAITING FOR FACES ---");
}

void loop() {
  // If HuskyLens isn't responding, don't crash, just wait
  if (!huskylens.request()) {
    delay(100);
    return;
  }

  if (huskylens.isLearned() && huskylens.available()) {
    HUSKYLENSResult result = huskylens.read();
    
    if (result.command == COMMAND_RETURN_BLOCK) {
      int faceID = result.ID;

      if (faceID != lastPlayedID) {
        // Stop any current audio before switching folders
        mp3.stop();
        delay(200);

        if (faceID == 1) {
          Serial.println("MATCH: Face ID 1 -> Playing Folder 01");
          mp3.play(1, 1); // Folder 01, File 1
          lastPlayedID = 1;
        } 
        else if (faceID == 2) {
          Serial.println("MATCH: Face ID 2 -> Playing Folder 02");
          mp3.play(2, 1); // Folder 02, File 1
          lastPlayedID = 2;
        }

        delay(3000); // Wait for audio to play
      }
    }
  } else {
    lastPlayedID = -1;
  }
}
